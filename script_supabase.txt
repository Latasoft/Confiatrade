-- 1. Tabla de usuarios (gestionados por Clerk)
CREATE TABLE IF NOT EXISTS users (
  id SERIAL PRIMARY KEY,
  clerk_id VARCHAR(255) UNIQUE NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  nombre VARCHAR(100),
  apellido VARCHAR(100),
  empresa VARCHAR(100),
  rol VARCHAR(20) NOT NULL DEFAULT 'cliente', -- 'cliente' o 'admin'
  pais VARCHAR(100),
  ciudad VARCHAR(100),
  direccion VARCHAR(255),
  telefono VARCHAR(50),
  fecha_registro TIMESTAMP DEFAULT NOW(),
  estado VARCHAR(20) DEFAULT 'activo'
);

ALTER TABLE users ENABLE ROW LEVEL SECURITY;

-- 2. Productos
CREATE TABLE IF NOT EXISTS productos (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  nombre VARCHAR(255) NOT NULL,
  descripcion TEXT,
  imagen_url TEXT,
  ubicacion VARCHAR(255),
  proveedor VARCHAR(255),
  stock INTEGER,
  unidad VARCHAR(50),
  precio NUMERIC(12,2),
  estado VARCHAR(20) DEFAULT 'disponible',
  fecha_creacion TIMESTAMP DEFAULT NOW()
);

ALTER TABLE productos ENABLE ROW LEVEL SECURITY;

-- 3. Transportes
CREATE TABLE IF NOT EXISTS transportes (
  id SERIAL PRIMARY KEY,
  nombre VARCHAR(255) NOT NULL,
  descripcion TEXT,
  imagen_url TEXT,
  origen VARCHAR(255),
  destino VARCHAR(255),
  capacidad_total NUMERIC(12,2),
  capacidad_disponible NUMERIC(12,2),
  unidad_capacidad VARCHAR(50),
  peso_maximo NUMERIC(12,2),
  transportista VARCHAR(255),
  vehiculo VARCHAR(255),
  tiempo_estimado VARCHAR(100),
  fecha_salida DATE,
  precio NUMERIC(12,2),
  estado VARCHAR(20) DEFAULT 'disponible',
  fecha_creacion TIMESTAMP DEFAULT NOW()
);

ALTER TABLE transportes ENABLE ROW LEVEL SECURITY;

-- 4. Pedidos
CREATE TABLE IF NOT EXISTS pedidos (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  producto_id INTEGER REFERENCES productos(id) ON DELETE SET NULL,
  cantidad NUMERIC(12,2),
  total NUMERIC(12,2),
  estado VARCHAR(20) DEFAULT 'pendiente',
  direccion_entrega VARCHAR(255),
  fecha_estimada DATE,
  notas TEXT,
  fecha_creacion TIMESTAMP DEFAULT NOW()
);

ALTER TABLE pedidos ENABLE ROW LEVEL SECURITY;

-- 5. Reservas de transporte
CREATE TABLE IF NOT EXISTS reservas (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  transporte_id INTEGER REFERENCES transportes(id) ON DELETE SET NULL,
  cantidad_reservada NUMERIC(12,2),
  estado VARCHAR(20) DEFAULT 'pendiente',
  fecha_reserva TIMESTAMP DEFAULT NOW(),
  notas TEXT
);

ALTER TABLE reservas ENABLE ROW LEVEL SECURITY;

-- 6. Reportes
CREATE TABLE IF NOT EXISTS reportes (
  id SERIAL PRIMARY KEY,
  tipo VARCHAR(20),
  fecha_generado TIMESTAMP DEFAULT NOW(),
  datos JSONB,
  generado_por INTEGER REFERENCES users(id) ON DELETE SET NULL
);

ALTER TABLE reportes ENABLE ROW LEVEL SECURITY;

-- 7. Pagos
CREATE TABLE IF NOT EXISTS pagos (
  id SERIAL PRIMARY KEY,
  pedido_id INTEGER REFERENCES pedidos(id) ON DELETE SET NULL,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  monto NUMERIC(12,2),
  moneda VARCHAR(10),
  metodo_pago VARCHAR(50),
  estado VARCHAR(20) DEFAULT 'pendiente',
  fecha_pago TIMESTAMP DEFAULT NOW(),
  referencia_externa VARCHAR(255)
);

ALTER TABLE pagos ENABLE ROW LEVEL SECURITY;

-- 8. Ventas
CREATE TABLE IF NOT EXISTS ventas (
  id SERIAL PRIMARY KEY,
  producto_id INTEGER REFERENCES productos(id) ON DELETE SET NULL,
  vendedor_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  comprador_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  pedido_id INTEGER REFERENCES pedidos(id) ON DELETE SET NULL,
  cantidad NUMERIC(12,2),
  total NUMERIC(12,2),
  fecha_venta TIMESTAMP DEFAULT NOW()
);

ALTER TABLE ventas ENABLE ROW LEVEL SECURITY;

-- 9. Métodos de pago
CREATE TABLE IF NOT EXISTS metodos_pago (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  tipo VARCHAR(50),
  datos JSONB,
  activo BOOLEAN DEFAULT TRUE
);

ALTER TABLE metodos_pago ENABLE ROW LEVEL SECURITY;

-- 10. Transacciones
CREATE TABLE IF NOT EXISTS transacciones (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  tipo VARCHAR(50),
  monto NUMERIC(12,2),
  moneda VARCHAR(10),
  descripcion TEXT,
  referencia_externa VARCHAR(255),
  fecha_transaccion TIMESTAMP DEFAULT NOW()
);

ALTER TABLE transacciones ENABLE ROW LEVEL SECURITY;

-- 11. Mensajes (chat)
CREATE TABLE IF NOT EXISTS mensajes (
  id SERIAL PRIMARY KEY,
  pedido_id INTEGER REFERENCES pedidos(id) ON DELETE SET NULL,
  emisor_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  receptor_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  mensaje TEXT,
  fecha_envio TIMESTAMP DEFAULT NOW()
);

ALTER TABLE mensajes ENABLE ROW LEVEL SECURITY;

-- 12. Reviews
CREATE TABLE IF NOT EXISTS reviews (
  id SERIAL PRIMARY KEY,
  producto_id INTEGER REFERENCES productos(id) ON DELETE SET NULL,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  calificacion INTEGER CHECK (calificacion >= 1 AND calificacion <= 5),
  comentario TEXT,
  fecha_review TIMESTAMP DEFAULT NOW()
);

ALTER TABLE reviews ENABLE ROW LEVEL SECURITY;

-- 13. Logs (auditoría)
CREATE TABLE IF NOT EXISTS logs (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
  accion VARCHAR(100),
  entidad VARCHAR(100),
  entidad_id INTEGER,
  descripcion TEXT,
  fecha_log TIMESTAMP DEFAULT NOW()
);

ALTER TABLE logs ENABLE ROW LEVEL SECURITY;

CREATE INDEX IF NOT EXISTS idx_transportes_origen_destino ON transportes(origen, destino);
CREATE INDEX IF NOT EXISTS idx_transportes_fecha_salida ON transportes(fecha_salida);
